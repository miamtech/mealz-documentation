## Start the Library

export function StartLibrary(props) {
  if (props.platform === "android") {
    return (
      <pre><code className="language-kotlin">{`import ai.mealz.uikit.config.SdkConfiguration
import ai.mealz.uikit.config.SdkContext
import ai.mealz.uikit.model.Environment

SdkConfiguration.start(
    supplierKey = WEB_SUPPLIER_KEY, // base64-encoded web key
    context = SdkContext(applicationContext),
    environment = Environment.PROD // or Environment.DEV
)`}</code></pre>
    );
  }
  if (props.platform === "ios") {
    return (
      <pre><code className="language-swift">{`import MealzUIKit

SdkConfiguration.shared.start(
    supplierKey: WEB_SUPPLIER_KEY, // base64-encoded web key
    context: SdkContext(),
    environment: .prod // or .dev
)`}</code></pre>
    );
  }
  return (
    <pre><code className="language-kotlin">{`import ai.mealz.uikit.config.SdkConfiguration
import ai.mealz.uikit.config.SdkContext
import ai.mealz.uikit.model.Environment

// From commonMain — pass a platform-provided SdkContext
SdkConfiguration.start(
    supplierKey = WEB_SUPPLIER_KEY, // base64-encoded web key
    context = sdkContext,
    environment = Environment.PROD // or Environment.DEV
)`}</code></pre>
  );
}

<StartLibrary platform={props.platform}/>

:::tip Supplier key
MealzUIKit expects the **web** supplier key format (base64 JSON including `supplierId`, `origin`, `domain`, `noSupplier`, …) — the same key family used by Mealz SSR. If you do not already have a web/SSR key, Mealz will provide one for your integration.
:::

## User

`setUser` and `getCurrentUser` are **suspending** — call them from a coroutine (Kotlin) or an async context that can await Kotlin suspend functions (Swift). Call after `start`, and again whenever auth changes.

### Authless Mode

**Default:** enabled (`isAuthlessModeEnabled()` is `true` until you call `disableAuthlessMode()`). Call `disableAuthlessMode()` when shoppers must log in before using Mealz.

While no retailer user id is set, the library treats the shopper as a guest (`User.Authless`) with a stable authless id. That lets them browse and fill a Mealz basket before login.

When you later call `setUser("logged-in-id")`:

1. If authless mode is **enabled** and the current user is still `User.Authless`, Mealz may **merge** the guest basket into the logged-in basket.
2. Merge runs only when a **store is already selected** (external id in retailer mode, or Mealz store id in No Supplier mode). Without a store, login still succeeds; merge is skipped.
3. Merge is **best-effort** (network failures are swallowed and do not block `setUser`).

Related SSR concepts: [Login and logout](/docs/web_ssr/v2/set-up-and-usage/login-and-logout).

export function OptionalConfig(props) {
  const isIos = props.platform === "ios";
  const langCode = isIos ? (
    <pre><code className="language-swift">{`SdkConfiguration.shared.setSelectedLanguage(language: .fr) // .en, …
SdkConfiguration.shared.getCurrentLanguage()`}</code></pre>
  ) : (
    <pre><code className="language-kotlin">{`SdkConfiguration.setSelectedLanguage(Language.FR) // Language.EN, …
SdkConfiguration.getCurrentLanguage()`}</code></pre>
  );
  const profilingCode = isIos ? (
    <pre><code className="language-swift">{`SdkConfiguration.shared.enableProfiling()
SdkConfiguration.shared.disableProfiling()`}</code></pre>
  ) : (
    <pre><code className="language-kotlin">{`SdkConfiguration.enableProfiling()
SdkConfiguration.disableProfiling()`}</code></pre>
  );
  const labelsCode = isIos ? (
    <pre><code className="language-swift">{`SdkConfiguration.shared.enableCustomLabels()
SdkConfiguration.shared.disableCustomLabels()`}</code></pre>
  ) : (
    <pre><code className="language-kotlin">{`SdkConfiguration.enableCustomLabels()
SdkConfiguration.disableCustomLabels()`}</code></pre>
  );
  const debugCode = isIos ? (
    <pre><code className="language-swift">{`SdkConfiguration.shared.enableDebugMode()
SdkConfiguration.shared.disableDebugMode()`}</code></pre>
  ) : (
    <pre><code className="language-kotlin">{`SdkConfiguration.enableDebugMode()
SdkConfiguration.disableDebugMode()`}</code></pre>
  );

  let debugBlurb = "Developer-only. Do not ship with this enabled in production.";
  if (props.platform === "android") {
    debugBlurb =
      "Developer-only. Enables Chrome WebView debugging (`chrome://inspect`). Do not ship with this enabled in production.";
  } else if (props.platform === "ios") {
    debugBlurb =
      "Developer-only. Makes the Mealz WKWebView inspectable in Safari. Do not ship with this enabled in production.";
  }

  const iosWebInspectorSteps =
    props.platform === "ios" ? (
      <ol>
        <li>
          On the iPhone: <strong>Settings → Safari → Advanced → Web Inspector</strong> → On.
        </li>
        <li>
          Call <code>SdkConfiguration.shared.enableDebugMode()</code> after <code>start</code>.
        </li>
        <li>
          On a Mac with the device connected (or simulator): Safari → <strong>Develop</strong> → your device / simulator →
          pick the Mealz WKWebView page.
        </li>
      </ol>
    ) : null;

  return (
    <>
      <h3>Language</h3>
      <p><strong>Default:</strong> <code>Language.FR</code> (<code>fr</code>).</p>
      {langCode}
      <p>
        Sets the locale used for SSR content (<code>Language-id</code>). Values map to ISO language codes (
        <code>fr</code>, <code>en</code>, …). See{' '}
        <a href="/docs/web_ssr/v2/customization/internationalization">SSR internationalization</a> for custom label files.
      </p>

      <h3>Profiling</h3>
      <p><strong>Default:</strong> enabled.</p>
      {profilingCode}
      <p>
        Controls personalization across Mealz features (recommendations, tailored catalog content, …). Mirror the shopper's
        consent: if profiling / cookies are refused, call <code>disableProfiling()</code>. This maps to the SSR{' '}
        <code>profiling</code> header (<code>true</code> / <code>false</code>).
      </p>

      <h3>Custom Labels</h3>
      <p><strong>Default:</strong> disabled.</p>
      {labelsCode}
      <p>
        When enabled, SSR uses your <strong>custom i18n override</strong> files (if configured with Mealz) instead of the
        default Mealz copy. Keep this off unless you have custom label packs. Details:{' '}
        <a href="/docs/web_ssr/v2/customization/internationalization">SSR internationalization</a>.
      </p>

      <h3>Debug Mode</h3>
      <p><strong>Default:</strong> disabled.</p>
      {debugCode}
      <p>{debugBlurb}</p>
      {iosWebInspectorSteps}
    </>
  );
}
