import Link from '@docusaurus/Link';

export function InstallContent(props) {
  if (props.platform === "android") {
    return (
      <>
        <p>MealzUIKit is published as <code>ai.mealz:ui-kit</code> (Maven Central).</p>
        <h2>Requirements</h2>
        <ul>
          <li><code>minSdk</code> <strong>24</strong></li>
        </ul>
        <h2>Gradle</h2>
        <pre><code className="language-gradle">{`// module build.gradle.kts
implementation("ai.mealz:ui-kit:1.0.0")`}</code></pre>
        <p>
          Latest published version for now: <strong>1.0.0</strong>. Use a newer patch/minor when Mealz asks you to upgrade.
        </p>
      </>
    );
  }
  if (props.platform === "ios") {
    return (
      <>
        <p>Distribute MealzUIKit via CocoaPods or Swift Package Manager from the release repository.</p>
        <h2>CocoaPods</h2>
        <pre><code className="language-ruby">{`pod 'MealzUIKit', '1.0.0'`}</code></pre>
        <h2>Swift Package Manager</h2>
        <p>Add the binary package from <a href="https://github.com/miamtech/MealzUIKitRelease">MealzUIKitRelease</a> (SPM product <code>MealzUIKit</code>).</p>
        <ul>
          <li>Release artifacts: <a href="https://github.com/miamtech/MealzUIKitRelease">https://github.com/miamtech/MealzUIKitRelease</a></li>
        </ul>
        <p>
          Latest published version for now: <strong>1.0.0</strong>. Requirements: iOS 12.0+, Swift 5.8+.
        </p>
        <h2>Info.plist (Required)</h2>
        <p>
          MealzUIKit embeds Compose Multiplatform UI. Add this key to your app <code>Info.plist</code>
          (without it the app can crash on iPhone):
        </p>
        <pre><code className="language-xml">{`&lt;key&gt;CADisableMinimumFrameDurationOnPhone&lt;/key&gt;
&lt;true/&gt;`}</code></pre>
      </>
    );
  }
  return (
    <>
      <p>From a Kotlin Multiplatform module, depend on the same artifact on all targets that use Mealz:</p>
      <pre><code className="language-kotlin">{`// shared/build.gradle.kts
kotlin {
    sourceSets {
        commonMain.dependencies {
            implementation("ai.mealz:ui-kit:1.0.0")
        }
    }
}`}</code></pre>
      <p>
        Latest published version for now: <strong>1.0.0</strong>. Android consumers need <code>minSdk</code> <strong>24</strong>.
        Use the shared Compose APIs (<code>CatalogScreen</code>, <code>RecipeCard</code>, …) from <code>commonMain</code>,
        and supply a platform <code>SdkContext</code> when calling <code>SdkConfiguration.start</code>.
      </p>
      <h2>iOS Info.plist (Required)</h2>
      <p>
        On the iOS app target, add this key to <code>Info.plist</code> (Compose Multiplatform requirement — without it the app can crash on iPhone):
      </p>
      <pre><code className="language-xml">{`&lt;key&gt;CADisableMinimumFrameDurationOnPhone&lt;/key&gt;
&lt;true/&gt;`}</code></pre>
    </>
  );
}

<InstallContent platform={props.platform}/>
